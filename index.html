Creating the Visual Studio Project

Step-by-step guide to creating your Class Library project with multi-version support.

1

Create a Class Library Project

Open Visual Studio → New Project → Class Library (.NET Framework)

  • Name: MyPlugin
  • Framework: .NET Framework 4.8
Important: Navisworks runs in-process. Your plugin is a .dll, not an .exe. Always use OutputType = Library.
2

Edit the .csproj for Multi-Version Support

Replace the default property groups with conditional ones. This lets you select a target Navisworks version from the build configuration dropdown.

XML MyPlugin.csproj 
<?xml version="1.0" encoding="utf-8"?>
<Project ToolsVersion="15.0"
         xmlns="http://schemas.microsoft.com/developer/msbuild/2003">
  <Import Project="$(MSBuildExtensionsPath)\$(MSBuildToolsVersion)\Microsoft.Common.props"
          Condition="Exists('...')" />

  <!-- Default settings -->
  <PropertyGroup>
    <Configuration Condition=" '$(Configuration)' == '' ">Debug2024</Configuration>
    <Platform Condition=" '$(Platform)' == '' ">x64</Platform>
    <ProjectGuid>{YOUR-GUID-HERE}</ProjectGuid>
    <OutputType>Library</OutputType>
    <RootNamespace>MyPlugin</RootNamespace>
    <AssemblyName>MyPlugin</AssemblyName>
    <UseWPF>true</UseWPF>
    <UseWindowsForms>true</UseWindowsForms>
    <TargetFrameworkVersion>v4.8</TargetFrameworkVersion>
  </PropertyGroup>

  <!-- Navisworks 2022 -->
  <PropertyGroup Condition="$(Configuration.Contains('2022'))">
    <NavisworksVersion>2022</NavisworksVersion>
    <TargetFrameworkVersion>v4.7.2</TargetFrameworkVersion>
    <OutputPath>bin\$(Configuration)\x64\</OutputPath>
    <PlatformTarget>x64</PlatformTarget>
  </PropertyGroup>

  <!-- Navisworks 2024 -->
  <PropertyGroup Condition="$(Configuration.Contains('2024'))">
    <NavisworksVersion>2024</NavisworksVersion>
    <TargetFrameworkVersion>v4.8</TargetFrameworkVersion>
    <OutputPath>bin\$(Configuration)\x64\</OutputPath>
    <PlatformTarget>x64</PlatformTarget>
  </PropertyGroup>

  <!-- Navisworks 2025 -->
  <PropertyGroup Condition="$(Configuration.Contains('2025'))">
    <NavisworksVersion>2025</NavisworksVersion>
    <TargetFrameworkVersion>v4.8</TargetFrameworkVersion>
    <OutputPath>bin\$(Configuration)\x64\</OutputPath>
    <PlatformTarget>x64</PlatformTarget>
  </PropertyGroup>

  <Import Project="$(MSBuildToolsPath)\Microsoft.CSharp.targets" />
</Project>
Tip: Add Build Configurations named Debug2024, Debug2025 etc. in Visual Studio via Build → Configuration Manager.

CommandHandlerPlugin

Primary Plugin

The most important plugin type. It wires ribbon buttons to C# methods. Every button click triggers ExecuteCommand with the command ID.

Required Attributes

Attribute Role
[PluginAttribute] Declares plugin name, developer ID, tooltip, display name
[RibbonLayoutAttribute] Points to the XAML file defining the ribbon layout
[RibbonTab] Declares ribbon tabs this plugin creates
[Command] Registers individual button command IDs
C# MyCommands.cs 
using Autodesk.Navisworks.Api.Plugins;
using Autodesk.Navisworks.Api;
using System.Windows.Forms;

namespace MyPlugin
{
    [PluginAttribute("MyPluginCommands",          // Unique plugin name
                     "MyCompany",                  // Developer ID
                     ToolTip = "My Navisworks Plugin",
                     DisplayName = "My Plugin")]
    [RibbonLayoutAttribute("MyPlugin.xaml")]
    [RibbonTab("ID_MyMainTab", DisplayName = "My Plugin")]
    [Command("ID_CMD_HelloWorld", CanToggle = true, DisplayName = "Hello World")]
    [Command("ID_CMD_ExportData", CanToggle = true, DisplayName = "Export Data")]
    public class MyCommands : CommandHandlerPlugin
    {
        public override int ExecuteCommand(string name, params string[] parameters)
        {
            try
            {
                switch (name)
                {
                    case "ID_CMD_HelloWorld":
                        MessageBox.Show("Hello from Navisworks Plugin!");
                        break;
                    case "ID_CMD_ExportData":
                        MyExporter.RunExport();
                        break;
                }
            }
            catch (Exception ex)
            {
                MessageBox.Show($"Error: {ex.Message}", "Plugin Error",
                    MessageBoxButtons.OK, MessageBoxIcon.Error);
            }
            return 0; // Must always return 0
        }
    }
}
Critical: Every ID_CMD_* string must be identical across three places: the [Command] attribute, the switch case, and the XAML button's Id.

AddInPlugin

Automation

Used when Navisworks is started in automation mode (e.g., from a script or command line). Enables calling the plugin from external processes.

C# MyAddin.cs 
using Autodesk.Navisworks.Api.Plugins;

namespace MyPlugin
{
    [PluginAttribute("MyAddin", "MyCompany",
                     ToolTip = "My Addin",
                     DisplayName = "My Addin")]
    [AddInPlugin(AddInLocation.None)]  // 'None' = no UI button
    internal class MyAddin : AddInPlugin
    {
        public override int Execute(params string[] parameters)
        {
            if (Application.IsAutomated && parameters.Length == 0)
                throw new InvalidOperationException("No parameters provided");

            switch (parameters[0])
            {
                case "ID_CMD_HelloWorld":
                    MyCommands_Impl.HelloWorld(false);
                    break;
                case "ID_CMD_ExportData":
                    MyCommands_Impl.ExportData(false);
                    break;
            }
            return 0;
        }
    }
}

// AddInLocation Options:
// AddInLocation.None    — Not accessible from UI
// AddInLocation.AddIn   — Appears in Add-Ins menu

EventWatcherPlugin

Lifecycle

Used to inject items into the File/Application menu and to run code when the plugin loads or unloads.

C# MyEventPlugin.cs 
using Autodesk.Navisworks.Api.Plugins;
using Autodesk.Windows;

namespace MyPlugin
{
    [Plugin("MyEventPlugin", "MyCompany",
            DisplayName = "My Event Plugin")]
    internal class MyEventPlugin : EventWatcherPlugin
    {
        public override void OnLoaded()
        {
            var appMenu = ComponentManager.ApplicationMenu;
            if (appMenu != null)
            {
                var menuItem = new ApplicationMenuItem
                {
                    Text = "My Custom Action",
                    Id = "ID_CMD_HelloWorld",
                    CommandHandler = new MyCommandHandler(),
                    IsActive = true,
                    IsVisible = true
                };
                appMenu.MenuContent.Items.Add(menuItem);
                appMenu.SaveState();
            }
        }

        public override void OnUnloading()
        {
            // Cleanup code here
        }
    }
}

The managed API is read-only. To write custom (user-defined) properties, you must use the COM API bridge.
C#COM Property Writer
using Autodesk.Navisworks.Api.Interop.ComApi;
using ComApi = Autodesk.Navisworks.Api.Interop.ComApi;

public static bool SetParameterValue(
    ModelItem item, string category,
    string attribute, dynamic value)
{
    // Get the COM state
    InwOpState10 oState = ComApiBridge.State;

    // Convert .NET ModelItem → COM path
    InwOaPath oPath = ComApiBridge.ToInwOaPath(item);

    // Get the GUI property node
    InwGUIPropertyNode2 propNode =
        (InwGUIPropertyNode2)oState.GetGUIPropertyNode(oPath, true);

    int index = 1;
    bool found = false;

    // Iterate existing user-defined categories
    foreach (InwGUIAttribute2 cat in propNode.GUIAttributes())
    {
        if (cat.ClassUserName.ToUpper() == category.ToUpper()
            && cat.UserDefined)
        {
            InwOaPropertyVec propVec = (InwOaPropertyVec)
                oState.ObjectFactory(
                    nwEObjectType.eObjectType_nwOaPropertyVec);

            foreach (InwOaProperty existing in cat.Properties())
            {
                InwOaProperty newProp = (InwOaProperty)
                    oState.ObjectFactory(
                        nwEObjectType.eObjectType_nwOaProperty);
                newProp.name = existing.name;
                newProp.value = existing.value;

                if (newProp.name == attribute)
                {
                    newProp.value = value;
                    found = true;
                }
                propVec.Properties().Add(newProp);
            }

            if (!found)
            {
                InwOaProperty addProp = (InwOaProperty)
                    oState.ObjectFactory(
                        nwEObjectType.eObjectType_nwOaProperty);
                addProp.name = attribute;
                addProp.value = value;
                propVec.Properties().Add(addProp);
            }

            propNode.SetUserDefined(
                index, cat.ClassUserName,
                cat.ClassName, propVec);
            return true;
        }
        index++;
    }

    // Category not found — create new
    InwOaPropertyVec newVec = (InwOaPropertyVec)
        oState.ObjectFactory(
            nwEObjectType.eObjectType_nwOaPropertyVec);
    InwOaProperty newProperty = (InwOaProperty)
        oState.ObjectFactory(
            nwEObjectType.eObjectType_nwOaProperty);
    newProperty.name = attribute;
    newProperty.UserName = attribute;
    newProperty.value = value;
    newVec.Properties().Add(newProperty);

    propNode.SetUserDefined(0, category, category, newVec);
    return true;
}
C#Model Traversal
// Access all models in the document
foreach (Model model in Application.ActiveDocument.Models)
{
    ModelItem root = model.RootItem;

    // Get all geometric leaf items
    var geometricItems = root.Descendants

// Recursive descent helper
public static List<ModelItem> GetAllLeafItems(ModelItem item)
{
    var result = new List<ModelItem>();
    foreach (ModelItem child in item.Children)
    {
        if (!child.Children.Any())
            result.Add(child);           // leaf node
        else
            result.AddRange(GetAllLeafItems(child));
    }
    return result;
}

Clash Detection API

Programmatically perform interference checks between element sets.

⚡

Hard

Objects physically overlap (intersecting geometry).

ClashTestType.Hard
📏

Clearance

Objects are within a specified distance of each other.

ClashTestType.Clearance
🔄

Duplicates

Identical overlapping geometry detection.

ClashTestType.Duplicates
C#Clash Detection Runner
using Autodesk.Navisworks.Api.Clash;
using Autodesk.Navisworks.Api;

public static IList<ClashResult> RunClashTest(
    string testName,
    ModelItemCollection selectionA,
    ModelItemCollection selectionB,
    ClashTestType testType = ClashTestType.Hard,
    double tolerance = 0.01)
{
    Document document = Application.ActiveDocument;
    DocumentClash documentClash = document.GetClash();
    DocumentClashTests clashTests = documentClash.TestsData;

    // Create a new clash test
    ClashTest newTest = new ClashTest();
    newTest.DisplayName = testName;
    newTest.SelectionA.Selection.CopyFrom(selectionA);
    newTest.SelectionB.Selection.CopyFrom(selectionB);
    newTest.TestType = testType;
    newTest.Tolerance = tolerance;

    // ⚠️ Known bug: must clear before adding
    clashTests.TestsClear();
    clashTests.TestsAddCopy(newTest);
    clashTests.TestsRunAllTests();

    // Re-read from document (required to get results)
    newTest = clashTests.Tests[0] as ClashTest;

    var results = new List<ClashResult>();
    foreach (SavedItem item in newTest.Children)
    {
        if (item is ClashResultGroup group)
            foreach (SavedItem gi in group.Children)
                if (gi is ClashResult cr) results.Add(cr);

        if (item is ClashResult result)
            results.Add(result);
    }
    return results;
}
Known API Bug: You must call TestsClear() before TestsAddCopy(). Without clearing, results may be unreliable.

Advanced API Features

Selection sets, viewpoints, color overrides, and bounding box geometry.

C#Selection Sets
// List all selection sets (recursive)
public static void ListAllSelectionSets()
{
    Document doc = Application.ActiveDocument;
    PrintSets(doc.SavedItems.Value);
}

private static void PrintSets(SavedItemCollection items)
{
    foreach (SavedItem item in items)
    {
        if (item is GroupItem folder)
        {
            Debug.WriteLine($"[Folder] {folder.DisplayName}");
            PrintSets(folder.Children);  // recurse
        }
        else if (item is SelectionSet selSet)
            Debug.WriteLine($"[SelectionSet] {selSet.DisplayName}");
        else if (item is SearchSet searchSet)
            Debug.WriteLine($"[SearchSet] {searchSet.DisplayName}");
    }
}

// Save current selection as a new set
public static void SaveCurrentSelectionAsSet(string name)
{
    Document doc = Application.ActiveDocument;
    ModelItemCollection selected = doc.CurrentSelection.SelectedItems;
    if (selected.Count == 0) return;

    SelectionSet newSet = new SelectionSet(selected);
    newSet.DisplayName = name;
    doc.SavedItems.Value.Add(newSet);
}
C#Viewpoints
// Navigate to a saved viewpoint by name
public static void GoToViewpoint(string viewpointName)
{
    Document doc = Application.ActiveDocument;
    foreach (SavedItem item in doc.SavedViewpoints.Value)
    {
        if (item is SavedViewpoint vp &&
            vp.DisplayName.Equals(viewpointName,
                StringComparison.OrdinalIgnoreCase))
        {
            doc.SavedViewpoints.CurrentSavedViewpoint = vp;
            return;
        }
    }
}

// Save the current camera as a new viewpoint
public static void SaveCurrentView(string name)
{
    Document doc = Application.ActiveDocument;
    SavedViewpoint newVp = new SavedViewpoint(
        doc.SavedViewpoints.CurrentViewpoint);
    newVp.DisplayName = name;
    doc.SavedViewpoints.Value.Add(newVp);
}

public static void ResetAllColors()
{
    Document doc = Application.ActiveDocument;

// Extract bounding box of a ModelItem
public static void PrintBoundingBox(ModelItem item)
{
    if (!item.HasGeometry) return;

    BoundingBox3D bb = item.BoundingBox();
    Point3D min = bb.Min;
    Point3D max = bb.Max;

    // Compute centre
    double cx = (min.X + max.X) / 2.0;
    double cy = (min.Y + max.Y) / 2.0;
    double cz = (min.Z + max.Z) / 2.0;

    // Compute dimensions
    double width  = max.X - min.X;
    double depth  = max.Y - min.Y;
    double height = max.Z - min.Z;
}

// Aggregate bounding box for a group of items
public static BoundingBox3D GetGroupBoundingBox(
    ModelItemCollection items)
{
    double minX = double.MaxValue, minY = double.MaxValue,
           minZ = double.MaxValue;
    double maxX = double.MinValue, maxY = double.MinValue,
           maxZ = double.MinValue;

    foreach (ModelItem item in items)
    {
        if (!item.HasGeometry) continue;
        BoundingBox3D bb = item.BoundingBox();
        if (bb.Min.X < minX) minX = bb.Min.X;
        if (bb.Min.Y < minY) minY = bb.Min.Y;
        if (bb.Min.Z < minZ) minZ = bb.Min.Z;
        if (bb.Max.X > maxX) maxX = bb.Max.X;
        if (bb.Max.Y > maxY) maxY = bb.Max.Y;
        if (bb.Max.Z > maxZ) maxZ = bb.Max.Z;
    }

    return new BoundingBox3D(
        new Point3D(minX, minY, minZ),
        new Point3D(maxX, maxY, maxZ));
}
